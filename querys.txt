CREATE TABLE vendas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    evento_id INT,
    nome_comprador VARCHAR(255),
    email VARCHAR(255),
    quantidade INT,
    data_venda DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (evento_id) REFERENCES eventos(id)
);
